# 05 外设编程与代码示例（Arduino 框架）

> 所有示例可直接复制进 Arduino IDE 编译。引脚按 [引脚分配建议](02-引脚图与安全使用.md)。

## 1. GPIO 输出：点灯与无阻塞闪烁

```cpp
const int LED = 2;
uint32_t prev = 0; bool on = false;
void setup() { pinMode(LED, OUTPUT); }
void loop() {
  if (millis() - prev >= 500) { prev = millis(); on = !on; digitalWrite(LED, on); }
}
```

## 2. 按键输入（内部上拉 + 消抖）

```cpp
const int BTN = 0;
bool last = HIGH, stable = HIGH; uint32_t t = 0;
void setup() { pinMode(BTN, INPUT_PULLUP); Serial.begin(115200); }
void loop() {
  bool r = digitalRead(BTN);
  if (r != last) { t = millis(); last = r; }
  if (millis() - t > 30 && r != stable) {
    stable = r;
    Serial.printf("按键 %s\n", stable == LOW ? "按下" : "松开");
  }
}
```

## 3. LEDC PWM：呼吸灯

```cpp
const int CH = 0, PIN = 25;
void setup() { ledcAttach(PIN, 5000, 8); }   // 5kHz, 8bit
void loop() {
  for (int d = 0; d <= 255; d++) { ledcWrite(PIN, d); delay(8); }
  for (int d = 255; d >= 0; d--) { ledcWrite(PIN, d); delay(8); }
}
```

## 4. ADC：读电位器 + 多次采样中位数

```cpp
const int PIN = 34;
uint16_t buf[16];
void setup() { Serial.begin(115200); analogSetAttenuation(ADC_11db); }
void loop() {
  for (int i = 0; i < 16; i++) buf[i] = analogRead(PIN);
  std::sort(buf, buf + 16);
  float v = buf[8] / 4095.0 * 3.3;
  Serial.printf("ADC=%d  V=%.3f\n", buf[8], v);
  delay(200);
}
```

## 5. 外部中断：脉冲计数

```cpp
volatile uint32_t cnt = 0;
void IRAM_ATTR isr() { cnt++; }
void setup() {
  Serial.begin(115200);
  pinMode(4, INPUT_PULLUP);
  attachInterrupt(4, isr, FALLING);
}
void loop() { delay(1000); Serial.printf("脉冲=%u\n", cnt); }
```

## 8. SPI：SD 卡列目录

```cpp
#include <SD.h>
#include <SPI.h>
void setup() {
  Serial.begin(115200);
  SPI.begin(18, 19, 23, 5);
  if (SD.begin(5)) {
    File root = SD.open("/");
    for (File f = root.openNextFile(); f; f = root.openNextFile())
      Serial.println(f.name());
  } else Serial.println("SD 初始化失败");
}
void loop() {}
```

## 9. WiFi 扫描 + Web 服务器（手机看传感器）

```cpp
#include <WiFi.h>
#include <WebServer.h>
const char* SSID = "your-wifi"; const char* PASS = "your-pass";
WebServer web(80);
void setup() {
  Serial.begin(115200);
  WiFi.begin(SSID, PASS);
  while (WiFi.status() != WL_CONNECTED) { delay(300); Serial.print("."); }
  Serial.printf("\nIP: %s\n", WiFi.localIP().toString().c_str());
  web.on("/", []() {
    web.send(200, "text/html",
      "<h1>ESP32</h1><p>ADC: " + String(analogRead(34)) +
      "</p><meta http-equiv='refresh' content='2'>");
  });
  web.begin();
}
void loop() { web.handleClient(); }
```

## 10. MQTT 上报（接入 Home Assistant/云）

```cpp
#include <WiFi.h>
#include <PubSubClient.h>
WiFiClient net; PubSubClient mqtt(net);
void setup() {
  Serial.begin(115200);
  WiFi.begin("ssid", "pass");
  while (WiFi.status() != WL_CONNECTED) delay(300);
  mqtt.setServer("192.168.1.10", 1883);
}
void loop() {
  if (!mqtt.connected()) mqtt.connect("esp32-node");
  mqtt.loop();
  static uint32_t t = 0;
  if (millis() - t > 5000) {
    t = millis();
    mqtt.publish("home/temp", String(random(200, 300) / 10.0).c_str());
  }
}
```

## 11. 深度睡眠 + 定时唤醒

```cpp
#define uS_TO_S(s) ((uint64_t)s * 1000000)
void setup() {
  Serial.begin(115200);
  Serial.println(" awake, work 5s then sleep 30s");
  delay(5000);
  esp_sleep_enable_timer_wakeup(uS_TO_S(30));
  esp_deep_sleep_start();
}
void loop() {}
```

## 12. BLE 广播（手机 nRF Connect 可见）

```cpp
#include <BLEDevice.h>
#include <BLEServer.h>
void setup() {
  BLEDevice::init("ESP32-DEMO");
  BLEServer* s = BLEDevice::createServer();
  BLEService* svc = s->createService("12345678-1234-1234-1234-1234567890ab");
  BLECharacteristic* c = svc->createCharacteristic("abcd", BLECharacteristic::PROPERTY_READ);
  c->setValue("hello ble");
  svc->start();
  s->getAdvertising()->start();
}
void loop() { delay(1000); }
```
